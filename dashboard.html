`; const message = document.getElementById("playerMessage"); const list = document.getElementById("playersList"); // Fetch exact registration counts async function loadPlayerCounts() { const [total, pending, approved, rejected] = await Promise.all([ db.from("registrations") .select("id", { count: "exact", head: true }), db.from("registrations") .select("id", { count: "exact", head: true }) .eq("registration_status", "pending"), db.from("registrations") .select("id", { count: "exact", head: true }) .eq("registration_status", "approved"), db.from("registrations") .select("id", { count: "exact", head: true }) .eq("registration_status", "rejected") ]); if (total.error || pending.error || approved.error || rejected.error) { message.textContent = "Count error: " + (total.error || pending.error || approved.error || rejected.error).message; return; } document.getElementById("totalPlayers").textContent = total.count ?? 0; document.getElementById("pendingPlayers").textContent = pending.count ?? 0; document.getElementById("approvedPlayers").textContent = approved.count ?? 0; document.getElementById("rejectedPlayers").textContent = rejected.count ?? 0; } // Fetch player records const { data, error } = await db .from("registrations") .select("*") .order("created_at", { ascending: false }); if (error) { message.textContent = "Error: " + error.message; return; } await loadPlayerCounts(); function renderPlayers() { const search = document.getElementById("playerSearch") .value.trim().toLowerCase(); const filter = document.getElementById("playerFilter").value; const filtered = data.filter(p => { const name = (p.player_name || "").toLowerCase(); const application = (p.application_no || "").toLowerCase(); const mobile = (p.mobile || "").toLowerCase(); return ( (!filter || p.registration_status === filter) && ( name.includes(search) || application.includes(search) || mobile.includes(search) ) ); }); message.textContent = filtered.length + " players found."; list.innerHTML = ""; if (filtered.length === 0) { list.textContent = "No players found."; return; } filtered.forEach(p => { const card = document.createElement("div"); card.className = "card"; card.style.marginTop = "12px"; const title = document.createElement("h3"); title.textContent = p.player_name || "Unnamed Player"; const details = document.createElement("p"); details.textContent = "Application: " + (p.application_no || "N/A") + " | Mobile: " + (p.mobile || "N/A") + " | Age: " + (p.age || "N/A") + " | Role: " + (p.role || "N/A") + " | Status: " + (p.registration_status || "N/A"); const view = document.createElement("button"); view.textContent = "View Details"; view.onclick = () => { alert( "Name: " + (p.player_name || "N/A") + "\nApplication: " + (p.application_no || "N/A") + "\nMobile: " + (p.mobile || "N/A") + "\nAge: " + (p.age || "N/A") + "\nAddress: " + (p.address || "N/A") + "\nRole: " + (p.role || "N/A") + "\nStatus: " + (p.registration_status || "N/A") ); }; card.append(title, details, view); ["approved", "rejected"].forEach(status => { const btn = document.createElement("button"); btn.textContent = status === "approved" ? "Approve" : "Reject"; btn.style.margin = "8px"; btn.onclick = async () => { const confirmed = confirm( "Are you sure you want to " + status + " this player?" ); if (!confirmed) return; btn.disabled = true; const { error } = await db .from("registrations") .update({ registration_status: status }) .eq("id", p.id); if (error) { alert("Update failed: " + error.message); btn.disabled = false; return; } alert("Player " + status + " successfully!"); await loadPlayers(); }; card.appendChild(btn); }); list.appendChild(card); }); } document.getElementById("playerSearch") .addEventListener("input", renderPlayers); document.getElementById("playerFilter") .addEventListener("change", renderPlayers); document.getElementById("refreshPlayers") .addEventListener("click", loadPlayers); renderPlayers(); } // Refresh player data every 30 seconds setInterval(() => { if ( document.visibilityState === "visible" && document.getElementById("playersSection") ) { loadPlayers(); } }, 30000);
